cmake_minimum_required(VERSION 3.20)
project(ChromaticMemoryRecycling VERSION 1.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Threads REQUIRED)
add_library(zero_core src/board.cpp src/network.cpp src/search.cpp src/proof.cpp src/telemetry_json.cpp)
target_include_directories(zero_core PUBLIC include)
if(MSVC)
  target_compile_options(zero_core PRIVATE /W4 /permissive- /fp:precise)
else()
  target_compile_options(zero_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(chromatic_depth src/main_mcts.cpp)
target_link_libraries(chromatic_depth PRIVATE zero_core Threads::Threads)
add_library(hybrid_core src/pvs.cpp src/rrn.cpp)
target_include_directories(hybrid_core PUBLIC include)
target_link_libraries(hybrid_core PUBLIC zero_core Threads::Threads)
if(MSVC)
  target_compile_options(hybrid_core PRIVATE /W4 /permissive- /fp:precise)
else()
  target_compile_options(hybrid_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(project_chromatic_hybrid src/main.cpp)
target_link_libraries(project_chromatic_hybrid PRIVATE hybrid_core zero_core Threads::Threads)
add_executable(depth_probe tools/depth_probe.cpp)
target_link_libraries(depth_probe PRIVATE zero_core Threads::Threads)
enable_testing()
add_executable(zero_tests tests/core.cpp)
target_link_libraries(zero_tests PRIVATE zero_core)
add_test(NAME core COMMAND zero_tests)
add_executable(depth_telemetry_tests tests/depth_telemetry.cpp)
target_link_libraries(depth_telemetry_tests PRIVATE zero_core)
add_test(NAME depth_uniform COMMAND depth_telemetry_tests)
add_test(NAME depth_deployed_model COMMAND depth_telemetry_tests "${CMAKE_CURRENT_SOURCE_DIR}/dist/default.zero")
add_executable(memory_recycling_tests tests/memory_recycling.cpp)
target_link_libraries(memory_recycling_tests PRIVATE zero_core)
add_test(NAME memory_uniform COMMAND memory_recycling_tests)
add_test(NAME memory_deployed_model COMMAND memory_recycling_tests "${CMAKE_CURRENT_SOURCE_DIR}/dist/default.zero")
add_executable(value_zero_tests tests/value_zero.cpp)
target_link_libraries(value_zero_tests PRIVATE zero_core Threads::Threads)
add_test(NAME value_zero_uniform COMMAND value_zero_tests)
add_test(NAME value_zero_deployed_model COMMAND value_zero_tests "${CMAKE_CURRENT_SOURCE_DIR}/dist/default.zero")
add_executable(hybrid_tests tests/hybrid_tests.cpp)
target_link_libraries(hybrid_tests PRIVATE hybrid_core Threads::Threads)
add_test(NAME hybrid_core COMMAND hybrid_tests)
option(CHROMATIC_ASAN "Enable AddressSanitizer for memory recycling validation" OFF)
if(CHROMATIC_ASAN)
  if(MSVC)
    target_compile_options(zero_core PUBLIC /fsanitize=address /Z7)
    target_link_options(zero_core INTERFACE /INCREMENTAL:NO)
    get_filename_component(CHROMATIC_COMPILER_DIR "${CMAKE_CXX_COMPILER}" DIRECTORY)
    foreach(target chromatic_depth project_chromatic_hybrid depth_probe zero_tests depth_telemetry_tests memory_recycling_tests value_zero_tests hybrid_tests)
      add_custom_command(TARGET ${target} POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
        "${CHROMATIC_COMPILER_DIR}/clang_rt.asan_dynamic-x86_64.dll" "$<TARGET_FILE_DIR:${target}>")
    endforeach()
  else()
    target_compile_options(zero_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(zero_core INTERFACE -fsanitize=address,undefined)
  endif()
endif()
add_library(pristine_core baseline/src/board.cpp baseline/src/network.cpp baseline/src/search.cpp baseline/src/proof.cpp)
target_include_directories(pristine_core PUBLIC baseline/include)
if(MSVC)
  target_compile_options(pristine_core PRIVATE /W4 /permissive- /fp:precise)
else()
  target_compile_options(pristine_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(pristine_probe tools/depth_probe.cpp)
target_compile_definitions(pristine_probe PRIVATE CHROMATIC_PRISTINE_PROBE)
target_link_libraries(pristine_probe PRIVATE pristine_core Threads::Threads)
