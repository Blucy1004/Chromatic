cmake_minimum_required(VERSION 3.20)
project(ChromaticDepthTelemetry VERSION 1.0.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Threads REQUIRED)
add_library(zero_core src/board.cpp src/network.cpp src/search.cpp src/proof.cpp src/telemetry_json.cpp)
target_include_directories(zero_core PUBLIC include)
if(MSVC)
  target_compile_options(zero_core PRIVATE /W4 /permissive- /fp:precise)
else()
  target_compile_options(zero_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(chromatic_depth src/main.cpp)
target_link_libraries(chromatic_depth PRIVATE zero_core Threads::Threads)
add_executable(depth_probe tools/depth_probe.cpp)
target_link_libraries(depth_probe PRIVATE zero_core Threads::Threads)
enable_testing()
add_executable(zero_tests tests/core.cpp)
target_link_libraries(zero_tests PRIVATE zero_core)
add_test(NAME core COMMAND zero_tests)
add_executable(depth_telemetry_tests tests/depth_telemetry.cpp)
target_link_libraries(depth_telemetry_tests PRIVATE zero_core)
add_test(NAME depth_uniform COMMAND depth_telemetry_tests)
add_test(NAME depth_deployed_model COMMAND depth_telemetry_tests "${CMAKE_CURRENT_SOURCE_DIR}/dist/default.zero")
add_library(pristine_core baseline/src/board.cpp baseline/src/network.cpp baseline/src/search.cpp baseline/src/proof.cpp)
target_include_directories(pristine_core PUBLIC baseline/include)
if(MSVC)
  target_compile_options(pristine_core PRIVATE /W4 /permissive- /fp:precise)
else()
  target_compile_options(pristine_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(pristine_probe tools/depth_probe.cpp)
target_compile_definitions(pristine_probe PRIVATE CHROMATIC_PRISTINE_PROBE)
target_link_libraries(pristine_probe PRIVATE pristine_core Threads::Threads)
