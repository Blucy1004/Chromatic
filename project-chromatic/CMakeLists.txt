cmake_minimum_required(VERSION 3.20)
project(Chromatic VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Threads REQUIRED)
add_library(zero_core src/board.cpp src/network.cpp src/search.cpp src/proof.cpp)
target_include_directories(zero_core PUBLIC include)
if(MSVC)
  target_compile_options(zero_core PRIVATE /W4 /permissive- /fp:precise)
else()
  target_compile_options(zero_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(project_zero src/main.cpp)
target_link_libraries(project_zero PRIVATE zero_core Threads::Threads)
enable_testing()
add_executable(zero_tests tests/core.cpp)
target_link_libraries(zero_tests PRIVATE zero_core)
add_test(NAME core COMMAND zero_tests)
option(CHROMATIC_ASAN "Build native targets with AddressSanitizer" OFF)
if(CHROMATIC_ASAN)
  if(MSVC)
    target_compile_options(zero_core PUBLIC /fsanitize=address /Z7)
    target_link_options(zero_core INTERFACE /INCREMENTAL:NO)
    get_filename_component(CHROMATIC_COMPILER_DIR "${CMAKE_CXX_COMPILER}" DIRECTORY)
    foreach(target project_zero zero_tests)
      add_custom_command(TARGET ${target} POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
        "${CHROMATIC_COMPILER_DIR}/clang_rt.asan_dynamic-x86_64.dll" "$<TARGET_FILE_DIR:${target}>")
    endforeach()
  else()
    target_compile_options(zero_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(zero_core INTERFACE -fsanitize=address,undefined)
  endif()
endif()
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/src/search.cpp" CHROMATIC_SEARCH_SOURCE)
string(REPLACE "value=-value;" "value=value;" CHROMATIC_MUTANT_SOURCE "${CHROMATIC_SEARCH_SOURCE}")
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/sign_mutant.cpp" CONTENT "${CHROMATIC_MUTANT_SOURCE}")
add_executable(sign_mutant EXCLUDE_FROM_ALL tests/core.cpp "${CMAKE_CURRENT_BINARY_DIR}/sign_mutant.cpp")
target_link_libraries(sign_mutant PRIVATE zero_core Threads::Threads)
add_library(baseline_core baseline/instrumented/src/board.cpp baseline/instrumented/src/network.cpp baseline/instrumented/src/search.cpp)
target_include_directories(baseline_core PUBLIC baseline/instrumented/include include)
add_executable(baseline_bench tools/native_bench.cpp)
target_compile_definitions(baseline_bench PRIVATE CHROMATIC_BASELINE)
target_link_libraries(baseline_bench PRIVATE baseline_core Threads::Threads)
if(WIN32)
  target_link_libraries(baseline_bench PRIVATE Psapi)
endif()
add_executable(chromatic_bench tools/native_bench.cpp)
target_link_libraries(chromatic_bench PRIVATE zero_core Threads::Threads)
if(WIN32)
  target_link_libraries(chromatic_bench PRIVATE Psapi)
endif()
