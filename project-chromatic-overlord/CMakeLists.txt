cmake_minimum_required(VERSION 3.20)
project(ChromaticOverlord VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Threads REQUIRED)
option(CHROMATIC_ASAN "Address and undefined behavior sanitizers" OFF)
option(CHROMATIC_TSAN "Thread sanitizer (separate build)" OFF)
option(CHROMATIC_WINDOWS_PORTABLE "Windows x64 static runtime and UTF-8 process manifest" OFF)
set(CHROMATIC_TEST_MODEL "${CMAKE_CURRENT_SOURCE_DIR}/../project-chromatic-memory/dist/default.zero" CACHE FILEPATH "Model used by native regression tests")
if(CHROMATIC_ASAN AND CHROMATIC_TSAN)
  message(FATAL_ERROR "Choose one sanitizer configuration")
endif()
if(CHROMATIC_WINDOWS_PORTABLE)
  if(NOT MSVC OR NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "The portable Windows package requires MSVC x64")
  endif()
  if(CHROMATIC_ASAN OR CHROMATIC_TSAN)
    message(FATAL_ERROR "Use a separate sanitizer build, not the portable package")
  endif()
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded")
endif()
add_library(zero_core src/board.cpp src/network.cpp src/search.cpp src/proof.cpp src/telemetry_json.cpp src/verification.cpp src/sha256.cpp)
target_include_directories(zero_core PUBLIC include)
target_link_libraries(zero_core PUBLIC Threads::Threads)
if(MSVC)
  target_compile_options(zero_core PRIVATE /W4 /permissive- /fp:precise)
  if(CHROMATIC_ASAN)
    target_compile_options(zero_core PUBLIC /fsanitize=address /Zi)
    target_link_options(zero_core INTERFACE /INCREMENTAL:NO)
  endif()
else()
  target_compile_options(zero_core PRIVATE -Wall -Wextra -Wpedantic)
  if(CHROMATIC_ASAN)
    target_compile_options(zero_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(zero_core PUBLIC -fsanitize=address,undefined)
  elseif(CHROMATIC_TSAN)
    target_compile_options(zero_core PUBLIC -fsanitize=thread -fno-omit-frame-pointer)
    target_link_options(zero_core PUBLIC -fsanitize=thread)
  endif()
endif()
add_executable(chromatic_overlord src/main.cpp)
target_link_libraries(chromatic_overlord PRIVATE zero_core)
if(CHROMATIC_WINDOWS_PORTABLE)
  target_link_options(chromatic_overlord PRIVATE /MANIFEST:EMBED "/MANIFESTINPUT:${CMAKE_CURRENT_SOURCE_DIR}/packaging/windows.manifest")
endif()
enable_testing()
foreach(name core depth_telemetry memory_recycling overlord model_identity research)
  add_executable(${name}_tests tests/${name}.cpp)
  target_link_libraries(${name}_tests PRIVATE zero_core)
  add_test(NAME ${name} COMMAND ${name}_tests)
  if(NOT name STREQUAL "core")
    add_test(NAME ${name}_model COMMAND ${name}_tests "${CHROMATIC_TEST_MODEL}")
  endif()
endforeach()
